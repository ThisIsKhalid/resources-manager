Functions must know 
// multiple return values (key Go idiom)
func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("division by zero")
    }
    return a / b, nil
}

// variadic
func sum(nums ...int) int {
    total := 0
    for _, n := range nums { total += n }
    return total
}

// first-class functions

Defer, panic, recover important 
func readFile() error {
    f, err := os.Open("data.txt")
    if err != nil { return err }
    defer f.Close()  // runs when function exits (LIFO)
    // ... work with f
    return nil
}

// panic = unrecoverable crash (avoid for normal errors)
func mustConnect(dsn string) *sql.DB {
    db, err := sql.Open("postgres", dsn)
    if err != nil { panic(err) }  // only at startup
    return db
}

// recover — catch a panic (e.g. in HTTP middleware)
defer func() {
    if r := recover(); r != nil {
        log.Printf("recovered: %v", r)
    }
}()

Error handling critical 
// errors are just values — handle explicitly
result, err := doWork()
if err != nil {
    return fmt.Errorf("doWork failed: %w", err) // wrap with %w
}

// custom error type
type NotFoundError struct { ID int }
func (e *NotFoundError) Error() string {
    return fmt.Sprintf("id %d not found", e.ID)
}

// unwrap with errors.Is / errors.As
var nfe *NotFoundError
if errors.As(err, &nfe) {
    // handle specifically
}

Types & structs

Go's type system — structs, interfaces, embedding

Struct definition & methods core 
type User struct {
    ID    int
    Name  string
    Email string
    role  string  // unexported (lowercase)
}

// value receiver — read only
func (u User) String() string {
    return fmt.Sprintf("%s <%s>", u.Name, u.Email)
}

// pointer receiver — can mutate
func (u *User) Promote(role string) {
    u.role = role
}

// constructor pattern (Go has no constructors)
func NewUser(name, email string) (*User, error) {
    if email == "" { return nil, errors.New("email required") }
    return &User{Name: name, Email: email}, nil
}
Interfaces critical for jobs 
// interface = implicit — no "implements" keyword
type Storer interface {
    Save(user User) error
    FindByID(id int) (User, error)
}

// concrete type satisfies the interface
type PostgresStore struct{ db *sql.DB }

func (p *PostgresStore) Save(u User) error { ... }
func (p *PostgresStore) FindByID(id int) (User, error) { ... }

// inject the interface, not the concrete type
type UserService struct { store Storer }

func NewUserService(s Storer) *UserService {
    return &UserService{store: s}
}

// now easy to mock in tests
type MockStore struct{}
func (m *MockStore) Save(u User) error { return nil }
func (m *MockStore) FindByID(id int) (User, error) { ... }
Embedding (composition over inheritance)
type Animal struct { Name string }
func (a Animal) Speak() string { return a.Name + " speaks" }

type Dog struct {
    Animal              // embedded — promotes all fields & methods
    Breed string
}

d := Dog{Animal: Animal{Name: "Rex"}, Breed: "Husky"}
d.Speak()    // promoted from Animal — "Rex speaks"
d.Name       // also promoted

// embed interfaces to compose larger interfaces
type ReadWriter interface {
    io.Reader
    io.Writer
}
Generics (Go 1.18+) modern 
// type parameter with constraint
func Map[T, U any](slice []T, fn func(T) U) []U {
    result := make([]U, len(slice))
    for i, v := range slice {
        result[i] = fn(v)
    }
    return result
}

names := Map(users, func(u User) string { return u.Name })

// constraint using ~
type Number interface { ~int | ~float64 }

func Sum[T Number](nums []T) T {
    var total T
    for _, n := range nums { total += n }
    return total
}

Concurrency

Go's superpower — goroutines, channels, sync primitives

Goroutines + channels must master 
// goroutine — cheap (2KB stack, not an OS thread)
go func() {
    fmt.Println("runs concurrently")
}()

// channel — typed pipe for communication
ch := make(chan int)       // unbuffered
buf := make(chan int, 10)  // buffered

go func() { ch <- 42 }()   // send
val := <-ch                 // receive (blocks)

// range over channel until closed
for v := range ch { fmt.Println(v) }

// directional channels (best practice in function sigs)
func produce(out chan<- int) { out <- 1 }
func consume(in <-chan int) { fmt.Println(<-in) }
Select + context (cancellation) must master 
// select — multiplex channels
select {
case msg := <-ch1:
    fmt.Println("ch1:", msg)
case msg := <-ch2:
    fmt.Println("ch2:", msg)
case <-time.After(5 * time.Second):
    fmt.Println("timeout")
default:
    // non-blocking
}

// context — pass through the call tree
ctx, cancel := context.WithTimeout(
    context.Background(), 5*time.Second,
)
defer cancel()

result, err := db.QueryContext(ctx, "SELECT ...")
if err != nil { ... }  // err is context.DeadlineExceeded on timeout
sync package important 
var mu sync.Mutex
mu.Lock()
defer mu.Unlock()

// RWMutex — multiple readers, single writer
var rw sync.RWMutex
rw.RLock() ; defer rw.RUnlock()  // read lock

// WaitGroup — wait for goroutines
var wg sync.WaitGroup
for _, item := range items {
    wg.Add(1)
    go func(i Item) {
        defer wg.Done()
        process(i)
    }(item)
}
wg.Wait()

// sync.Once — run exactly once (e.g. singleton)
var once sync.Once
once.Do(func() { db = connectDB() })
sync/atomic + errgroup pro tools 
// atomic — lock-free counter
var count atomic.Int64
count.Add(1)
fmt.Println(count.Load())

// errgroup — goroutines that return errors
import "golang.org/x/sync/errgroup"

g, ctx := errgroup.WithContext(ctx)
g.Go(func() error { return fetchUsers(ctx) })
g.Go(func() error { return fetchOrders(ctx) })

if err := g.Wait(); err != nil {
    return err  // first error from any goroutine
}

Layer architecture 
// handler → service → repo → db

handler: HTTP, validation,
         request/response

service: business logic,
         orchestration

repository: DB queries,
            returns domain types

domain: plain structs,
        no framework deps
HTTP server (net/http + chi) standard 
func main() {
    db := connectDB()
    repo := repository.NewUserRepo(db)
    svc  := service.NewUserService(repo)
    h    := handler.NewUserHandler(svc)

    r := chi.NewRouter()
    r.Use(middleware.Logger, middleware.Recoverer)
    r.Use(middleware.Timeout(30 * time.Second))

    r.Route("/users", func(r chi.Router) {
        r.Get("/", h.List)
        r.Post("/", h.Create)
        r.Get("/{id}", h.Get)
    })

    srv := &http.Server{
        Addr:         ":8080",
        Handler:      r,
        ReadTimeout:  10 * time.Second,
        WriteTimeout: 10 * time.Second,
        IdleTimeout:  60 * time.Second,
    }
    log.Fatal(srv.ListenAndServe())
}
Graceful shutdown production must 
quit := make(chan os.Signal, 1)
signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
<-quit  // block until signal

ctx, cancel := context.WithTimeout(
    context.Background(), 30*time.Second,
)
defer cancel()

if err := srv.Shutdown(ctx); err != nil {
    log.Fatal("forced shutdown:", err)
}
log.Println("server exited cleanly")
Testing always write tests 
// table-driven test (Go standard)
func TestDivide(t *testing.T) {
    tests := []struct{
        name    string
        a, b    float64
        want    float64
        wantErr bool
    }{
        {"normal", 10, 2, 5, false},
        {"by zero", 10, 0, 0, true},
    }
    for _, tc := range tests {
        t.Run(tc.name, func(t *testing.T) {
            got, err := Divide(tc.a, tc.b)
            if (err != nil) != tc.wantErr { t.Fatalf(...) }
            if got != tc.want { t.Errorf(...) }
        })
    }
}

Job roadmap

From zero to Go engineer — with what to build

1
Week 1–2 — Language fundamentals
Syntax, slices, maps, structs, interfaces, error handling. Read "A Tour of Go" (tour.golang.org). Build: CLI tool — a task manager reading/writing JSON to disk.
2
Week 3–4 — Concurrency deep dive
Goroutines, channels, select, WaitGroup, Mutex, context. Read: "Concurrency in Go" (Katherine Cox-Buday). Build: concurrent web scraper or URL checker.
3
Month 2 — HTTP API + database
net/http, chi/gin/echo, sqlx or pgx, migrations (goose/migrate). Build: REST API with auth (JWT), CRUD, and PostgreSQL. Deploy to Railway or Fly.io.
4
Month 3 — Production-grade skills
Structured logging (slog), graceful shutdown, config (envconfig/viper), Docker, Makefile. Add middleware: rate limiting, request ID, CORS. Write table-driven tests, benchmarks.
5
Month 4 — Scalability topics
gRPC + protobuf, Redis caching, message queues (NATS/Kafka basics), worker pools. Build: microservice or background job processor. Learn: profiling with pprof.
6
Month 5–6 — Interview prep
LeetCode in Go (medium tier: trees, graphs, dp). System design for Go services. Go-specific interview topics: goroutine leaks, channel deadlocks, GC, scheduler (GOMAXPROCS).

Three Things That Will Set You Apart in Go Interviews

  • Interface-Driven Design & Dependency Injection — Build testable, maintainable applications by coding against interfaces rather than concrete implementations.
  • Goroutine Lifecycle Management — Prevent leaks and handle cancellations correctly using context.Context.
  • Modern Error Handling — Use error wrapping with %w and inspect errors properly with errors.As() and errors.Is().
